Tablas Relevantes para el cálculo del saldo presupuestario
CREATE TABLE `presupuestos` (
	`ID_Presupuesto` INT NOT NULL AUTO_INCREMENT,
	`origen_de_financiamiento_id_of` INT NOT NULL COMMENT 'origen de financiamiento del presupuesto',
	`fuente_de_financiamiento_id_ff` INT NOT NULL COMMENT 'Fuente de fianciamiento del presupuesto',
	`programa_id_pro` INT NOT NULL,
	`Idcuentacontable` INT NOT NULL,
	`Año` DATE NULL DEFAULT NULL,
	`TotalPresupuestado` DECIMAL(15,0) NOT NULL,
	`TotalModificado` DECIMAL(15,0) NULL DEFAULT NULL,
	`id_uni_respon_usu` INT NOT NULL,
	`estado` INT NOT NULL,
	PRIMARY KEY (`ID_Presupuesto`) USING BTREE,
	INDEX `origen_de_financiamiento_id_of` (`origen_de_financiamiento_id_of`) USING BTREE,
	INDEX `fuente_de_financiamiento_id_ff` (`fuente_de_financiamiento_id_ff`) USING BTREE,
	INDEX `programa_id_pro` (`programa_id_pro`) USING BTREE,
	INDEX `Idcuentacontable` (`Idcuentacontable`) USING BTREE,
	INDEX `id_uni_respon_usu` (`id_uni_respon_usu`) USING BTREE,
	CONSTRAINT `cc_fk` FOREIGN KEY (`Idcuentacontable`) REFERENCES `cuentacontable` (`IDCuentaContable`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `ff_fk` FOREIGN KEY (`fuente_de_financiamiento_id_ff`) REFERENCES `fuente_de_financiamiento` (`id_ff`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `id_cc` FOREIGN KEY (`Idcuentacontable`) REFERENCES `cuentacontable` (`IDCuentaContable`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `id_uni_respon_usu` FOREIGN KEY (`id_uni_respon_usu`) REFERENCES `uni_respon_usu` (`id_uni_respon_usu`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `of_fk` FOREIGN KEY (`origen_de_financiamiento_id_of`) REFERENCES `origen_de_financiamiento` (`id_of`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `pr_fk` FOREIGN KEY (`programa_id_pro`) REFERENCES `programa` (`id_pro`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
;
CREATE TABLE `presupuesto_mensual` (
	`id_presupuesto_mes` INT NOT NULL AUTO_INCREMENT COMMENT 'identificador del plan financiero o  tambien llamado presupuesto mensual',
	`id_presupuesto` INT NOT NULL COMMENT 'el presupuesto que percibe el plan financiero',
	`mes` DATETIME NOT NULL COMMENT 'separar siempre el mes de este campo',
	`monto_presupuestado` DECIMAL(15,0) NOT NULL COMMENT 'Monto del presupuesto del mes',
	`monto_modificado` DECIMAL(15,0) NULL DEFAULT '0' COMMENT 'este es el mismo monto que el presupuesto gral, ignorar aquí ',
	PRIMARY KEY (`id_presupuesto_mes`) USING BTREE,
	INDEX `idx_presupuesto_mes` (`id_presupuesto`, `mes`) USING BTREE,
	INDEX `idx_mes_presupuesto` (`mes`) USING BTREE,
	INDEX `idx_montos_presupuesto` (`monto_presupuestado`, `monto_modificado`) USING BTREE,
	CONSTRAINT `presupuesto_mensual_ibfk_1` FOREIGN KEY (`id_presupuesto`) REFERENCES `presupuestos` (`ID_Presupuesto`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
;

	`str` INT NOT NULL,

	INDEX `id_usuario_numasi` (`id_usuario_numasi`) USING BTREE,
	INDEX `id_nivel_str` (`id_nivel_str`) USING BTREE,
	CONSTRAINT `fk_id_nivel_str` FOREIGN KEY (`id_nivel_str`) REFERENCES `nivel` (`id_nivel`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_id_unidad_responsable` FOREIGN KEY (`id_uni_respon_usu`) REFERENCES `unidad_academica` (`id_unidad`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_id_usuario_numasi` FOREIGN KEY (`id_usuario_numasi`) REFERENCES `usuarios` (`id_user`) ON UPDATE NO ACTION ON DELETE NO ACTION

;

CREATE TABLE `ejecucion_mensual` (
	`id_ejecucion_mes` INT NOT NULL AUTO_INCREMENT,
	`id_presupuesto` INT NOT NULL COMMENT 'el presupuesto que recibe este presupuesto mensual',
	`mes` DATETIME NOT NULL COMMENT 'mes al que aplicará el presupuesto ',
	`obligado` DECIMAL(15,0) NULL DEFAULT '0' COMMENT 'monto obligado en el mes (acá se sumael monto de las obligaciones )',
	`pagado` DECIMAL(15,0) NULL DEFAULT '0' COMMENT 'monto pagado en el mes (acá se suma el monto de los debe)',
	`actualizado_en` TIMESTAMP NULL DEFAULT (CURRENT_TIMESTAMP) ON UPDATE CURRENT_TIMESTAMP,
	`creado_en` TIMESTAMP NULL DEFAULT (CURRENT_TIMESTAMP),
	PRIMARY KEY (`id_ejecucion_mes`) USING BTREE,
	UNIQUE INDEX `unique_mes_ejecucion` (`id_presupuesto`, `mes`) USING BTREE,
	CONSTRAINT `ejecucion_mensual_ibfk_1` FOREIGN KEY (`id_presupuesto`) REFERENCES `presupuestos` (`ID_Presupuesto`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=10
;
CREATE TABLE `comprobante_gasto` (
	`IDComprobanteGasto` INT NOT NULL AUTO_INCREMENT,
	`id_pedido` INT NOT NULL COMMENT 'Un pedido agrupa todos los items que se compraron en un comprobante de gasto',
	`id_unidad` INT NOT NULL COMMENT 'unidad académica que pertenece el usuario que hizo un comprobante de gasto',
	`fecha` DATE NOT NULL,
	`idproveedor` INT NULL DEFAULT NULL COMMENT 'proveedor asociado al pago del comprobante de gasto ',
	`idpresupuesto` INT NOT NULL COMMENT 'presupuesto que se usará en el comprobante de gasto, se debe verificar primero que el presupuesto_mensual asociado a este presupuesto tenga saldo disponible en su presupuesto mesual ',
	`descripcion` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_general_ci',
	`concepto` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_general_ci',
	`id_item` INT NOT NULL,
	`preciounit` INT NOT NULL,
	`cantidad` INT NOT NULL,
	`iva` INT NOT NULL,
	`porcentaje_iva` INT NOT NULL,
	`exenta` INT NOT NULL,
	`gravada` INT NOT NULL,
	`id_uni_respon_usu` INT NOT NULL,
	`estado` INT NOT NULL,
	`procesado_asiento` TINYINT(1) NULL DEFAULT '0' COMMENT 'Indica si el comprobante ya fue procesado en un asiento contable',
	`numero_asiento` INT NULL DEFAULT NULL COMMENT 'Referencia al asiento contable generado',
	`fecha_procesado` DATETIME NULL DEFAULT NULL COMMENT 'Fecha cuando se procesó el asiento',
	PRIMARY KEY (`IDComprobanteGasto`) USING BTREE,
	INDEX `id_unidad` (`id_unidad`) USING BTREE,
	INDEX `id_uni_respon_usu` (`id_uni_respon_usu`) USING BTREE,
	INDEX `idproveedor` (`idproveedor`) USING BTREE,
	INDEX `idx_procesado` (`procesado_asiento`) USING BTREE,
	INDEX `idx_presupuesto_fecha` (`idpresupuesto`, `fecha`) USING BTREE,
	CONSTRAINT `fk_comprobante_proveedor` FOREIGN KEY (`idproveedor`) REFERENCES `proveedores` (`id`) ON UPDATE NO ACTION ON DELETE NO ACTION,
	CONSTRAINT `fk_comprobante_unidad` FOREIGN KEY (`id_unidad`) REFERENCES `unidad_academica` (`id_unidad`) ON UPDATE NO ACTION ON DELETE NO ACTION
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=50
;
